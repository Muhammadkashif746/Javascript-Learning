1. for Loop

for(let i = 1; i <= 10; i++) {
    console.log(i);
}

2. while Loop

let i = 1;
while(i <= 5) {
    console.log(i);
    i++;
}

3. do-while Loop

let i = 1;
do {
    console.log(i);
    i++;
} while(i <= 5);
