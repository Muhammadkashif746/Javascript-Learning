Data Types in JavaScript

← Back to Home

1. Primitive Data Types

String

let name = "Ahmed";
let greeting = 'Hello World';
let message = `My name is ${name}`;  // Template literal (ES6)

Number

let age = 20;
let price = 99.99;
let temperature = -5;

Boolean

let isStudent = true;
let isLoggedIn = false;

2. Special Values

let undefinedVar;           // undefined
let nullVar = null;            // null (intentional empty)

console.log(typeof undefinedVar); // "undefined"
console.log(typeof nullVar);      // "object" (famous bug)

3. Complex Data Types

// Object
let student = {
    name: "Sara",
    age: 21,
    isActive: true
};

// Array
let fruits = ["Apple", "Banana", "Mango"];

// Function
function add(a, b) {
    return a + b;
}
